CREATE DATABASE HackathonDB;
GO
USE HackathonDB;
GO

CREATE TABLE Usuarios (
    Id INT IDENTITY PRIMARY KEY,
    Nome NVARCHAR(150) NOT NULL,
    Email NVARCHAR(200) NOT NULL UNIQUE,
    Senha NVARCHAR(300) NOT NULL,
    Perfil VARCHAR(20) NOT NULL CHECK (Perfil IN ('ALUNO','CRIADOR'))
);

CREATE TABLE Eventos (
    Id INT IDENTITY PRIMARY KEY,
    CriadorId INT NOT NULL FOREIGN KEY REFERENCES Usuarios(Id),
    Titulo NVARCHAR(200) NOT NULL,
    Descricao NVARCHAR(MAX) NULL,
    DataHora DATETIME NOT NULL,
    Local NVARCHAR(200) NULL,
    Publicado BIT NOT NULL DEFAULT 1
);

CREATE TABLE Inscricoes (
    Id INT IDENTITY PRIMARY KEY,
    EventoId INT NOT NULL FOREIGN KEY REFERENCES Eventos(Id),
    AlunoId INT NOT NULL FOREIGN KEY REFERENCES Usuarios(Id),
    Presente BIT NOT NULL DEFAULT 0,
    DataInscricao DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT UQ_Evento_Aluno UNIQUE (EventoId, AlunoId)
);

CREATE TABLE Comentarios (
    Id INT IDENTITY PRIMARY KEY,
    EventoId INT NOT NULL FOREIGN KEY REFERENCES Eventos(Id),
    UsuarioId INT NOT NULL FOREIGN KEY REFERENCES Usuarios(Id),
    Texto NVARCHAR(1000) NOT NULL,
    DataComentario DATETIME NOT NULL DEFAULT GETDATE()
);
